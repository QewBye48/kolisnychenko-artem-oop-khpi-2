  + getSportType(): String
  + getLocation(): String

}

  - specialization: String

  + getFullName(): String
  + getSpecialization(): String
  + getExperienceYears(): int
  + toString(): String
}

  + Student()
  + Student(fullName: String, ticketId: String, studyYear: int)
  + getFullName(): String
  + getTicketId(): String
  + getStudyYear(): int

}

  - dayOfWeek: String
  - startTime: String
  - durationMinutes: int

  + getDayOfWeek(): String
  + getStartTime(): String
  + getDurationMinutes(): int

}
